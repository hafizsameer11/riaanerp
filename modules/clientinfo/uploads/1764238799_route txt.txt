SELECT 'Hide' as Hide, i.SellerCode, s.ID, s.Name AS Seller, i.ProductCode AS Code, i.ProductDescription AS Description , MIN(i.Date) StartDate, MAX(i.Date) EndDate, us.Name as Manufacturer 
FROM sc_transaction_import AS i 
INNER JOIN sc_seller AS s ON (s.Code=i.SellerCode) 
LEFT JOIN sc_product_matching AS pm ON ((i.ProductDescription=pm.ProductDescription) AND (i.ProductCode=pm.ProductCode) AND (s.ID=pm.ScSellerID)) 
LEFT JOIN sc_unknown_products_manufacturer upm on i.ProductDescription = upm.ProductDescription AND i.ProductCode = upm.ProductCode AND i.SellerCode = upm.SellerCode 
left join sc_seller us on upm.ScSellerID = us.id 
WHERE (pm.ProductCode IS NULL) 
GROUP BY s.ID, s.Name, i.ProductCode, i.ProductDescription ORDER BY i.Date, i.ProductCode, i.ProductDescription
